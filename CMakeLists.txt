cmake_minimum_required(VERSION 3.18)
include(pico_sdk_import.cmake)
project(littlecore C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(littlecore src/main.cpp src/music.cpp src/performance_controls.cpp src/space_effect.cpp)
target_include_directories(littlecore PRIVATE src)
target_link_libraries(littlecore pico_stdlib hardware_adc hardware_pwm hardware_clocks hardware_irq hardware_sync)
pico_enable_stdio_usb(littlecore 0)
pico_enable_stdio_uart(littlecore 0)
pico_add_extra_outputs(littlecore)
